# Function to parse bootstrap values from contree file
def parse_bootstrap_values(contree_file):
    """Parse bootstrap and SH-aLRT values from consensus tree"""
    if os.path.exists(contree_file):
        with open(contree_file, 'r') as f:
            tree = f.read().strip()
        # Extract numbers following branch separators
        # Pattern matches numbers in parentheses like )100/85 or )100 or )95/78
        matches = re.findall(r'\)([\d]+(?:/[\d]+)?)', tree)
        return matches
    return []

# Generate iTOL link
def generate_itol_link(tree_file, output_prefix="phylogenetic_tree"):
    """Generate iTOL upload link from tree file"""
    if os.path.exists(tree_file):
        import base64
        with open(tree_file, 'r') as f:
            tree_content = f.read()
        
        # Encode tree for iTOL URL
        tree_encoded = base64.b64encode(tree_content.encode()).decode()
        
        # Create direct iTOL link
        itol_link = f"https://itol.embl.de/upload.cgi?tree={tree_encoded}"
        
        # Also generate a shorter alternative link
        itol_alt = "https://itol.embl.de/"
        
        return itol_link, itol_alt
    return None, None
