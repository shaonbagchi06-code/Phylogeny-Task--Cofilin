import os
from google.colab import files


!pip install clipkit -q


print(" Upload your FASTA file:")
uploaded = files.upload()
filename = list(uploaded.keys())[0]


simple_name = "input.fasta"
import shutil
shutil.copy(filename, simple_name)

print(f"\n Processing: {simple_name}")


print("\n Trimming alignment...")
exit_code = os.system(f"clipkit {simple_name} -o trimmed.fasta -m smart-gap 2>&1")

if exit_code == 0:
    print(" Trimming successful!")
    
   
    if os.path.exists("trimmed.fasta"):
        print("\n Downloading trimmed alignment...")
        files.download("trimmed.fasta")
        
        
        print("\n Preview of trimmed alignment:")
        !head -n 10 trimmed.fasta
    else:
        print(" Output file not found")
else:
    print(f"\n ClipKIT failed with exit code: {exit_code}")
    

    print("\n Trying alternative command...")
    !python -m clipkit {simple_name} -o trimmed.fasta -m smart-gap
    
    if os.path.exists("trimmed.fasta"):
        print("Success with alternative command!")
        files.download("trimmed.fasta")
